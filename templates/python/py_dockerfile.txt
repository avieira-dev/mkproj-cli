# Stage 1
FROM python:3.10.21-slim

# Sets the working directory inside the container
WORKDIR /app

ENV PYTHONDONTWRITEBYTECODE=1
ENV PYTHONUNBUFFERED=1

# It copies the contents of the directory where you run 'docker build' to '/app' inside the image
COPY . ./

CMD ["python3", "src/$default_directory/main.py"]
