# Stage 1: Build
FROM node AS builder

WORKDIR /app

COPY package*.json .

RUN npm install

COPY tsconfig.json .
COPY src/ ./src

RUN npm run build

# Stage 2: Production image
FROM node

WORKDIR /app

COPY package*.json .

RUN npm install --omit=dev

COPY --from=builder /app/dist ./dist

CMD ["node", "dist/index.js"]
